import cv2
import numpy as np

# ==============================
# CONFIGURACIÓN
# ==============================

CAMARA = 1

camara = cv2.VideoCapture(CAMARA)

if not camara.isOpened():
    print("ERROR: No se pudo abrir la webcam USB.")
    exit()

print("Detector de colores iniciado.")
print("Colores: VERDE, AMARILLO, GRIS y NEGRO")
print("Presiona Q para salir.")


# ==============================
# FUNCIÓN PARA DETECTAR COLOR
# ==============================

def detectar_color(frame, mascara, nombre, color_rectangulo):

    contornos, _ = cv2.findContours(
        mascara,
        cv2.RETR_EXTERNAL,
        cv2.CHAIN_APPROX_SIMPLE
    )

    for contorno in contornos:

        area = cv2.contourArea(contorno)

        # Ignorar pequeñas manchas
        if area > 1000:

            x, y, w, h = cv2.boundingRect(contorno)

            # Dibujar rectángulo
            cv2.rectangle(
                frame,
                (x, y),
                (x + w, y + h),
                color_rectangulo,
                3
            )

            # Mostrar nombre
            cv2.putText(
                frame,
                nombre,
                (x, y - 10),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.8,
                color_rectangulo,
                2
            )


# ==============================
# BUCLE PRINCIPAL
# ==============================

while True:

    ret, frame = camara.read()

    if not ret:
        print("ERROR: No se pudo capturar la imagen.")
        break

    # ==============================
    # CONVERTIR BGR → HSV
    # ==============================

    hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)


    # ==============================
    # VERDE
    # ==============================

    verde_bajo = np.array([35, 70, 50])
    verde_alto = np.array([85, 255, 255])

    mascara_verde = cv2.inRange(
        hsv,
        verde_bajo,
        verde_alto
    )


    # ==============================
    # AMARILLO
    # ==============================

    amarillo_bajo = np.array([20, 70, 50])
    amarillo_alto = np.array([35, 255, 255])

    mascara_amarillo = cv2.inRange(
        hsv,
        amarillo_bajo,
        amarillo_alto
    )


    # ==============================
    # NEGRO
    # ==============================

    negro_bajo = np.array([0, 0, 0])
    negro_alto = np.array([180, 255, 80])

    mascara_negro = cv2.inRange(
        hsv,
        negro_bajo,
        negro_alto
    )


    # ==============================
    # GRIS
    # ==============================

    gris_bajo = np.array([0, 0, 81])
    gris_alto = np.array([180, 60, 200])

    mascara_gris = cv2.inRange(
        hsv,
        gris_bajo,
        gris_alto
    )


    # ==============================
    # LIMPIAR MÁSCARAS
    # ==============================

    kernel = np.ones((5, 5), np.uint8)

    mascara_verde = cv2.morphologyEx(
        mascara_verde,
        cv2.MORPH_OPEN,
        kernel
    )

    mascara_verde = cv2.morphologyEx(
        mascara_verde,
        cv2.MORPH_CLOSE,
        kernel
    )


    mascara_amarillo = cv2.morphologyEx(
        mascara_amarillo,
        cv2.MORPH_OPEN,
        kernel
    )

    mascara_amarillo = cv2.morphologyEx(
        mascara_amarillo,
        cv2.MORPH_CLOSE,
        kernel
    )


    mascara_negro = cv2.morphologyEx(
        mascara_negro,
        cv2.MORPH_OPEN,
        kernel
    )

    mascara_negro = cv2.morphologyEx(
        mascara_negro,
        cv2.MORPH_CLOSE,
        kernel
    )


    mascara_gris = cv2.morphologyEx(
        mascara_gris,
        cv2.MORPH_OPEN,
        kernel
    )

    mascara_gris = cv2.morphologyEx(
        mascara_gris,
        cv2.MORPH_CLOSE,
        kernel
    )


    # ==============================
    # DETECTAR VERDE
    # ==============================

    detectar_color(
        frame,
        mascara_verde,
        "VERDE",
        (0, 255, 0)
    )


    # ==============================
    # DETECTAR AMARILLO
    # ==============================

    detectar_color(
        frame,
        mascara_amarillo,
        "AMARILLO",
        (0, 255, 255)
    )


    # ==============================
    # DETECTAR NEGRO
    # ==============================

    detectar_color(
        frame,
        mascara_negro,
        "NEGRO",
        (255, 255, 255)
    )


    # ==============================
    # DETECTAR GRIS
    # ==============================

    detectar_color(
        frame,
        mascara_gris,
        "GRIS",
        (128, 128, 128)
    )


    # ==============================
    # MOSTRAR IMAGEN
    # ==============================

    cv2.imshow(
        "Detector de colores",
        frame
    )


    # ==============================
    # SALIR CON Q
    # ==============================

    if cv2.waitKey(1) & 0xFF == ord("q"):
        break


# ==============================
# CERRAR
# ==============================

camara.release()
cv2.destroyAllWindows()
